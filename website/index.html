Create an ESP32 WiFi scanner sketch that prints all nearby SSIDs and RSSI values every 5 seconds.

Here is your complete ESP32 WiFi scanner sketch:

#include <WiFi.h>

void setup() {
    Serial.begin(115200);
    WiFi.mode(WIFI_STA);
    WiFi.disconnect();
    delay(100);
}

void loop() {
    int n = WiFi.scanNetworks();
    Serial.println("Scan done!");
    for (int i = 0; i < n; ++i) {
        Serial.printf("%d: %s (%d dBm)\n", i + 1, WiFi.SSID(i).c_str(), WiFi.RSSI(i));
    }
    delay(5000);
}

Build from Source

Run locally in 4 simple command steps.

bash — quickstart 
# 1. Clone repository
git clone https://github.com/sanir321/arduino-ai.git
cd arduino-ai

# 2. Install dependencies & CLI binaries
yarn install

# 3. Build extension & electron app
yarn --cwd arduino-ide-extension build
yarn --cwd electron-app build:dev

# 4. Launch the IDE in development mode
yarn start
